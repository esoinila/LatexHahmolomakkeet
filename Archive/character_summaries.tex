\section*{Character Summaries}

\subsection*{Aziz (@DesertTombGuide)}
\begin{itemize}
    \item \textbf{Summary:} A charismatic tomb guide and YouTube personality using his platform to fund a secret war to restore his family's name after they were unjustly disgraced and their historical artifacts seized.
    \item \textbf{Personal Quest:} To find the legendary "Sunstone of Amun," the artifact his family was accused of stealing, and use it to clear their name.
    \item \textbf{Key Rival:} Fahim Al-Jabri, a corrupt Ministry official who orchestrated his family's downfall.
\end{itemize}

\subsection*{Jonathan Reed (@ArchaeologyDebunked)}
\begin{itemize}
    \item \textbf{Summary:} An Oxford postgraduate and YouTuber dedicated to exposing archaeological hoaxes after a rival discredited his early research.
    \item \textbf{Personal Quest:} To publicly expose the popular pseudo-archaeologist "Dr. Enigma" by infiltrating his inner circle and revealing his fraudulent methods.
    \item \textbf{Key Contact:} His mentor, Professor Alistair Finch at Oxford.
\end{itemize}

\subsection*{Field Technician Karl (@DesertTechGuru)}
\begin{itemize}
    \item \textbf{Summary:} A skilled field technician for Zenith Wave Technology, haunted by the death of his younger brother, Leo, in an accident involving salvaged tech.
    \item \textbf{Personal Quest:} To find the schematics for the pre-collapse device that killed his brother to understand what went wrong.
    \item \textbf{Key Contact:} A desert scavenger named "Ratchet," who provides him with rare parts and information.
\end{itemize}

\subsection*{Lead Designer Kate (@TeslaTechExplained)}
\begin{itemize}
    \item \textbf{Summary:} A brilliant Zenith Wave designer, wracked with guilt over a failed experiment that led to the disappearance of her mentor, Dr. Elias Vance.
    \item \textbf{Personal Quest:} To recover the classified data logs from the failed field test to find answers about her mentor's disappearance.
    \item \textbf{Key Contact:} Lena, her missing mentor's daughter, who provides cryptic clues.
\end{itemize}

\subsection*{Nova Cleo (@CleoOnlyFans)}
\begin{itemize}
    \item \textbf{Summary:} An OnlyFans star and influencer who uses her platform to fund expeditions to ancient sites, seeking to understand visions of a past life as an Atlantean priestess.
    \item \textbf{Personal Quest:} To locate the "Chamber of Echoes," a hidden chamber from her visions that she believes holds the key to her past self.
    \item \textbf{Key Rival:} A skeptical journalist, Ben Carter, who is trying to expose her as a fraud.
\end{itemize}

\subsection*{Orion Sagebrook (@ThirdEyeTransit)}
\begin{itemize}
    \item \textbf{Summary:} A mystic and Instagram influencer who maps the world's spiritual ley lines, driven by the grief of his partner's mysterious disappearance during a psychedelic ritual.
    \item \textbf{Personal Quest:} To gather ritual components from three different continents to perform a "Calling" ritual to bring his partner back.
    \item \textbf{Key Contact:} A reclusive mystic known as "The Weaver," who communicates only through riddles.
\end{itemize}

\subsection*{Sapphire Skye Monroe (@sapphireontherise)}
\begin{itemize}
    \item \textbf{Summary:} A TikTok influencer who plays the part of a clueless "Spiritual CEO" to secretly investigate the controversial theories of her disgraced academic parents.
    \item \textbf{Personal Quest:} To find the "Atlantean Star-Chart" her parents were searching for to vindicate their name.
    \item \textbf{Key Contact:} Dr. Miles Corbin, her parents' loyal but disgraced colleague.
\end{itemize}

\subsection*{Dr. Varn (@AncientTechOracle)}
\begin{itemize}
    \item \textbf{Summary:} A fringe theorist and YouTuber who abandoned mainstream astrophysics after witnessing his father's disappearance during a strange celestial event.
    \item \textbf{Personal Quest:} To find a device capable of predicting the next celestial alignment to prevent a similar event or find his father.
    \item \textbf{Key Rival:} Dr. Aris Thorne, a mainstream scientist who debunks his work.
\end{itemize}

\subsection*{Dr. Alaric Voss (@TheOracleOfStone)}
\begin{itemize}
    \item \textbf{Summary:} A disgraced Cambridge scholar and YouTuber whose controversial theories on Atlantis cost him his career and family.
    \item \textbf{Personal Quest:} To find the "Fountain of Life," a mythical Atlantean site, to cure his estranged wife, Eleonora.
    \item \textbf{Key Contact:} His estranged wife, Eleonora, who suffers from a rare degenerative disease.
\end{itemize}
