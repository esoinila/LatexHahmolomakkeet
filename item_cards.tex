% Card style using tcolorbox (Item)
\newcommand{\itemcard}[4]{%
    \begin{tcolorbox}[
        width=5.8cm, % Adjust width to fit in minipage
        colback=boxcontentbg,
        colframe=boxborder,
        coltitle=boxheaderfg,
        colbacktitle=boxheaderbg,
        fonttitle=\bfseries\sffamily\large,
        title=#1, % Item name as title
        arc=10pt, % Rounded corners
        boxrule=2pt, % Thick border
        left=3mm,
        right=3mm,
        top=2mm,
        bottom=2mm,
        enhanced,
        nobeforeafter,
    ]%
        % Content (Description, Usage, etc.)
        \textbf{Description:} #2\\
        \vspace{0.2cm}
        \textbf{Usage:} #3\\
        \vspace{0.3cm} % Space before flavor text
        \centering\itshape "#4"% Flavor text
    \end{tcolorbox}%
}%
