% Card style using tcolorbox
\newcommand{\itemguidecard}[3]{%
    \begin{tcolorbox}[
        width=8.5cm, % Width for two columns on A4
        height=5.5cm, % Fixed height for consistency
        colback=boxcontentbg,
        colframe=boxborder,
        coltitle=boxheaderfg,
        colbacktitle=boxheaderbg,
        fonttitle=\bfseries\sffamily\large,
        title=#1, % Item name
        arc=5pt,
        boxrule=1pt,
        left=3mm,
        right=3mm,
        top=2mm,
        bottom=2mm,
        enhanced,
        nobeforeafter,
        valign=top % Align content to the top
    ]
        \textbf{Description:} #2\par
        \vspace{0.2cm}
        \textbf{Usage:} #3
    \end{tcolorbox}%
}
