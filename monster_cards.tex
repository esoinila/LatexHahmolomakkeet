% Card style using tcolorbox (Monster)
\newcommand{\monstercard}[8]{%
    \begin{tcolorbox}[
        width=5.8cm, % Adjust width to fit in minipage
        colback=boxcontentbg,
        colframe=boxborder,
        coltitle=boxheaderfg,
        colbacktitle=boxheaderbg,
        fonttitle=\bfseries\sffamily\large,
        title=#1, % Monster name as title
        arc=10pt, % Rounded corners
        boxrule=2pt, % Thick border
        left=3mm,
        right=3mm,
        top=2mm,
        bottom=2mm,
        enhanced,
        nobeforeafter,
    ]%
        % Content (Type, HP, Abilities, etc.)
        \textbf{Type:} #2\\
        \textbf{HP:} #3\\
        \vspace{0.1cm} % Reduced space before abilities
        \textbf{Abilities:}
        \begin{itemize}[nosep, leftmargin=*, itemsep=0pt, topsep=0pt, parsep=0pt, partopsep=0pt]
            #4
        \end{itemize}
        \vspace{0.1cm} % Reduced space before weaknesses
        \textbf{Weaknesses:}
        \begin{itemize}[nosep, leftmargin=*, itemsep=0pt, topsep=0pt, parsep=0pt, partopsep=0pt]
            #5
        \end{itemize}
         \vspace{0.1cm} % Add space before resistances
        \textbf{Resistances:}
        \begin{itemize}[nosep, leftmargin=*, itemsep=0pt, topsep=0pt, parsep=0pt, partopsep=0pt]
            #6
        \end{itemize}
         \vspace{0.1cm} % Add space before special
        \textbf{Special:} #7
        
        \vspace{0.3cm} % Space before slogan
        \centering\itshape "#8"% Slogan
    \end{tcolorbox}%
}%
