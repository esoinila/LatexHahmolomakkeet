% Card style
\newcommand{\monstercard}[8]{
    \begin{tikzpicture}
        % Card background
        \fill[cardbg] (0,0) rectangle (6,9);
        % Title bar
        \fill[titlebg] (0,8) rectangle (6,9);
        \node[text=titletext, font=\bfseries] at (3,8.5) {#1};
        
        % Content
        \node[align=left, text width=5.5cm] at (3,7) {
            \textbf{Type:} #2\\
            \textbf{HP:} #3\\
            \textbf{Abilities:} #4\\
            \textbf{Weaknesses:} #5\\
            \textbf{Resistances:} #6\\
            \textbf{Special:} #7
        };
        
        % Slogan
        \node[align=center, text width=5.5cm, font=\itshape] at (3,2) {
            "#8"
        };
    \end{tikzpicture}
}
